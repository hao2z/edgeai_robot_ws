cmake_minimum_required(VERSION 3.8)
project(edgeai_mapping)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(edgeai_mapping)
find_package(nlohmann_json REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread)
find_package(TBB REQUIRED)
find_library(EDGEAI_ASYNC_SLAM async_slam_toolbox REQUIRED)
find_library(EDGEAI_SLAM_COMMON toolbox_common REQUIRED)
find_library(EDGEAI_KARTO kartoSlamToolbox REQUIRED)
add_executable(anchored_slam_toolbox src/anchored_slam_toolbox.cpp)
target_compile_features(anchored_slam_toolbox PRIVATE cxx_std_17)
ament_target_dependencies(anchored_slam_toolbox slam_toolbox std_msgs std_srvs)
target_link_libraries(anchored_slam_toolbox nlohmann_json::nlohmann_json
  ${EDGEAI_ASYNC_SLAM} ${EDGEAI_SLAM_COMMON} ${EDGEAI_KARTO} Boost::serialization Boost::thread TBB::tbb)
install(TARGETS anchored_slam_toolbox DESTINATION lib/${PROJECT_NAME})
add_executable(joint_slam_toolbox src/joint_slam_toolbox.cpp)
target_compile_features(joint_slam_toolbox PRIVATE cxx_std_17)
ament_target_dependencies(joint_slam_toolbox slam_toolbox std_msgs std_srvs)
target_link_libraries(joint_slam_toolbox nlohmann_json::nlohmann_json
  ${EDGEAI_SLAM_COMMON} ${EDGEAI_KARTO} Boost::serialization Boost::thread TBB::tbb)
install(TARGETS joint_slam_toolbox DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch scripts DESTINATION share/${PROJECT_NAME})
ament_package()
