cmake_minimum_required(VERSION 3.8)
project(edgeai_mission)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "action/ExecuteTask.action")
rosidl_get_typesupport_target(task_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(mission_coordinator src/mission_coordinator.cpp)
target_compile_features(mission_coordinator PRIVATE cxx_std_17)
ament_target_dependencies(mission_coordinator rclcpp rclcpp_action behaviortree_cpp nav_msgs std_msgs std_srvs geometry_msgs tf2 tf2_ros tf2_geometry_msgs)
target_link_libraries(mission_coordinator nlohmann_json::nlohmann_json "${task_typesupport}")
add_executable(task_executor src/task_executor.cpp)
target_compile_features(task_executor PRIVATE cxx_std_17)
ament_target_dependencies(task_executor rclcpp rclcpp_action behaviortree_cpp std_srvs)
target_link_libraries(task_executor nlohmann_json::nlohmann_json "${task_typesupport}")
install(TARGETS mission_coordinator DESTINATION lib/${PROJECT_NAME})
install(TARGETS task_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config scripts DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
