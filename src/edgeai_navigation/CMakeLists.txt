cmake_minimum_required(VERSION 3.8)
project(edgeai_navigation)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav2_rotation_shim_controller REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav2_behaviors REQUIRED)
add_library(occupied_escape SHARED src/occupied_escape.cpp)
ament_target_dependencies(occupied_escape nav2_behaviors nav2_msgs nav2_costmap_2d tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core behavior_plugins.xml)
install(TARGETS occupied_escape LIBRARY DESTINATION lib)
add_library(matched_rpp SHARED src/matched_rpp.cpp)
ament_target_dependencies(matched_rpp nav2_regulated_pure_pursuit_controller nav2_util std_msgs)
target_link_libraries(matched_rpp nlohmann_json::nlohmann_json)
install(TARGETS matched_rpp LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
add_library(fractional_window_layer SHARED src/fractional_window_layer.cpp)
ament_target_dependencies(fractional_window_layer nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_costmap_2d costmap_plugins.xml)
install(TARGETS fractional_window_layer LIBRARY DESTINATION lib)
add_library(xy_lattice_planner SHARED src/xy_lattice_planner.cpp)
ament_target_dependencies(xy_lattice_planner nav2_core nav2_util pluginlib tf2_geometry_msgs)
target_compile_features(xy_lattice_planner PUBLIC cxx_std_17)
target_link_libraries(xy_lattice_planner nlohmann_json::nlohmann_json)
pluginlib_export_plugin_description_file(nav2_core planner_plugins.xml)
install(TARGETS xy_lattice_planner LIBRARY DESTINATION lib)
add_library(edgeai_navigation_bt_nodes SHARED src/navigation_bt_nodes.cpp)
target_compile_features(edgeai_navigation_bt_nodes PUBLIC cxx_std_17)
target_compile_definitions(edgeai_navigation_bt_nodes PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(edgeai_navigation_bt_nodes nav2_behavior_tree behaviortree_cpp nav2_costmap_2d nav2_msgs tf2_geometry_msgs std_msgs)
target_link_libraries(edgeai_navigation_bt_nodes nlohmann_json::nlohmann_json)
install(TARGETS edgeai_navigation_bt_nodes LIBRARY DESTINATION lib)
install(DIRECTORY config launch scripts DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_navigation_heading ../../tests/exploration/test_navigation_heading.cpp)
  target_include_directories(test_navigation_heading PRIVATE src)
  ament_target_dependencies(test_navigation_heading nav2_core nav2_util nav2_costmap_2d tf2_geometry_msgs)
  ament_add_gtest(test_recovery_geometry ../../tests/exploration/test_recovery_geometry.cpp)
  target_include_directories(test_recovery_geometry PRIVATE src)
  target_compile_features(test_recovery_geometry PUBLIC cxx_std_17)
  ament_target_dependencies(test_recovery_geometry nav2_costmap_2d)
  ament_add_gtest(test_navigation_bt_nodes ../../tests/exploration/test_navigation_bt_nodes.cpp)
  target_compile_features(test_navigation_bt_nodes PUBLIC cxx_std_17)
  target_compile_definitions(test_navigation_bt_nodes PRIVATE TEST_PLUGIN_PATH="$<TARGET_FILE:edgeai_navigation_bt_nodes>")
  get_filename_component(NAV2_BT_PREFIX "${nav2_behavior_tree_DIR}/../../.." ABSOLUTE)
  target_compile_definitions(test_navigation_bt_nodes PRIVATE TEST_NAV2_BT_DIR="${NAV2_BT_PREFIX}/lib")
  ament_target_dependencies(test_navigation_bt_nodes behaviortree_cpp nav2_behavior_tree nav2_msgs tf2_geometry_msgs)
  add_dependencies(test_navigation_bt_nodes edgeai_navigation_bt_nodes)
endif()
ament_package()
